def _hover_card(e: ft.ControlEvent) -> None:
    active = e.data == "true"
    e.control.bgcolor = BG_CARD_HOVER if active else BG_CARD
    e.control.border = ft.border.all(1, BORDER_HOVER if active else BORDER)
    e.control.update()


def _hover_primary(e: ft.ControlEvent) -> None:
    e.control.bgcolor = ACCENT_HOVER if e.data == "true" else ACCENT
    e.control.update()


def _hover_ghost(e: ft.ControlEvent) -> None:
    active = e.data == "true"
    e.control.bgcolor = BG_CARD_HOVER if active else "transparent"
    e.control.border = ft.border.all(1, BORDER_HOVER if active else BORDER)
    e.control.update()


def _hover_danger(e: ft.ControlEvent) -> None:
    active = e.data == "true"
    e.control.bgcolor = "#3a1414" if active else "transparent"
    e.control.border = ft.border.all(1, DANGER if active else BORDER)
    e.control.update()


def _hover_icon(e: ft.ControlEvent) -> None:
    e.control.bgcolor = BG_CARD_HOVER if e.data == "true" else None
    e.control.update()


def _hover_row(e: ft.ControlEvent) -> None:
    e.control.bgcolor = BG_CARD_HOVER if e.data == "true" else None
    e.control.update()


def card(content, padding=SP_MD, hover=True, expand=False, on_click=None):
    c = ft.Container(content=content, bgcolor=BG_CARD,
                     border=ft.border.all(1, BORDER),

                     expand=expand, on_click=on_click, ink=False)

def confirm_dialog(page, title, message, on_confirm,
                   confirm_text="تأكيد", cancel_text="إلغاء",
                   danger=False):
    def close(e=None):
        page.close(dlg)

    def confirm(e):
        page.close(dlg)
        on_confirm()

    dlg = ft.AlertDialog(
        modal=True, bgcolor=BG_CARD,
        title=ft.Text(title, color=TEXT_PRIMARY,
                      weight=ft.FontWeight.W_700, size=FS_SECTION),
        content=ft.Text(message, color=TEXT_SECONDARY, size=FS_BODY),
        actions=[ghost_button(cancel_text, close, width=120),
                 ghost_button(confirm_text, confirm, width=120,
                              color=DANGER if danger else BG_BLACK,
                              danger=danger)])
    page.open(dlg)


def show_row_context_menu(page, edit_fn=None, delete_fn=None,
                          extra_items=None):
    def close(e=None):
        page.close(dlg)

    def do_edit(e):
        page.close(dlg)
        if edit_fn:
            edit_fn()

    def do_delete(e):
        page.close(dlg)
        if delete_fn:
            delete_fn()

    actions = []
    if edit_fn:
        actions.append(ghost_button("تعديل", do_edit,
                                    icon=ft.icons.EDIT, width=130))
    if delete_fn:
        actions.append(ghost_button("حذف", do_delete,
                                    icon=ft.icons.DELETE, width=130,
                                    color=DANGER, danger=True))
    if extra_items:
        actions.extend(extra_items)
    actions.append(ghost_button("إغلاق", close, width=110))

    dlg = ft.AlertDialog(
        modal=True, bgcolor=BG_CARD,
        title=ft.Text("خيارات السجل", color=TEXT_PRIMARY,
                      weight=ft.FontWeight.W_700, size=FS_SECTION),
        content=ft.Text("اختر إجراءً:", color=TEXT_SECONDARY,
                        size=FS_SMALL),
        actions=actions)
    page.open(dlg)


def data_table(page, columns, rows, empty_message="لا توجد سجلات.",
               context_actions=None):
    if not rows:
        return empty_state(empty_message)
    header = ft.Row(
        [ft.Container(ft.Text(h, color=TEXT_SECONDARY, size=FS_TINY,
                              weight=ft.FontWeight.W_700), width=w)
         for h, w in columns], spacing=SP_MD)
    body_rows = []
    for idx, r in enumerate(rows):
        cells = [ft.Container(c, width=columns[i][1],
                              alignment=ft.alignment.center_left)
                 for i, c in enumerate(r)]
        row = ft.Container(
            content=ft.Row(cells, spacing=SP_MD,
                           vertical_alignment=ft.CrossAxisAlignment.CENTER),
            padding=ft.padding.symmetric(vertical=8, horizontal=SP_MD),
            border=ft.border.only(bottom=ft.BorderSide(1, BORDER)))
        row.on_hover = _hover_row
        if context_actions and idx < len(context_actions) \
                and context_actions[idx]:
            e_fn, d_fn = context_actions[idx]
            if e_fn or d_fn:
                row.on_secondary_tap = (
                    lambda e, ef=e_fn, df=d_fn:
                    show_row_context_menu(page, ef, df))
        body_rows.append(row)
    return ft.Column(
        [ft.Container(content=header,
                      padding=ft.padding.symmetric(vertical=8,
                                                   horizontal=SP_MD),
                      border=ft.border.only(bottom=ft.BorderSide(1, BORDER))),
         ft.Column(body_rows, spacing=0, scroll=ft.ScrollMode.AUTO,
                   expand=True)],
        spacing=0, expand=True)

@contextmanager
def get_connection():
    conn = sqlite3.connect(DB_PATH, timeout=30.0)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON;")
    conn.execute("PRAGMA busy_timeout = 30000;")
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()

def _money(value):

def _row_to_dict(row):

def _rows_to_dicts(rows):
    return [dict(r) for r in rows]


def _date_filter(column, date_from=None, date_to=None):
    clauses, params = [], []
    if date_from:
        clauses.append(f"date({column}) >= date(?)")
        params.append(date_from)
    if date_to:
        clauses.append(f"date({column}) <= date(?)")
        params.append(date_to)
    return clauses, params


def hash_password(password, salt=None):
    salt = salt or secrets.token_hex(16)
    digest = hashlib.pbkdf2_hmac("sha256", str(password).encode("utf-8"),
                                 salt.encode("utf-8"), 120_000).hex()
    return f"{salt}${digest}"


def verify_password(password, stored_hash):
    if not stored_hash or "$" not in stored_hash:
        return False
    salt, _ = stored_hash.split("$", 1)
    return secrets.compare_digest(hash_password(password, salt),
                                  stored_hash)


SCHEMA_SQL = """
CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT NOT NULL UNIQUE,
    password TEXT NOT NULL,
    role TEXT NOT NULL DEFAULT 'user' CHECK (role IN ('admin', 'user')),
    created_at TEXT NOT NULL DEFAULT (datetime('now', 'localtime'))
);

    id INTEGER PRIMARY KEY AUTOINCREMENT,

    created_at TEXT NOT NULL DEFAULT (datetime('now', 'localtime'))

CREATE INDEX IF NOT EXISTS idx_products_supplier ON products (supplier);
CREATE TABLE IF NOT EXISTS suppliers (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,

    address TEXT NOT NULL DEFAULT '',
    notes TEXT NOT NULL DEFAULT '',
    created_at TEXT NOT NULL DEFAULT (datetime('now', 'localtime'))

    created_at TEXT NOT NULL DEFAULT (datetime('now', 'localtime')),

CREATE INDEX IF NOT EXISTS idx_invoices_created_at ON invoices (created_at);
CREATE TABLE IF NOT EXISTS invoice_items (
    id INTEGER PRIMARY KEY AUTOINCREMENT,

def initialize_database():
    os.makedirs(os.path.dirname(DB_PATH) or ".", exist_ok=True)
    os.makedirs(BACKUP_DIR, exist_ok=True)
    with get_connection() as conn:
        conn.execute("PRAGMA journal_mode = WAL;")
        conn.executescript(SCHEMA_SQL)
        # migrations
        for stmt in (
            "ALTER TABLE products ADD COLUMN supplier TEXT NOT NULL DEFAULT ''",
            "ALTER TABLE invoices ADD COLUMN invoice_number INTEGER",
        ):
            try:
                conn.execute(stmt)
            except sqlite3.OperationalError:
                pass
        # seed seasons
        for name in SEASON_NAMES:
            conn.execute(
                "INSERT OR IGNORE INTO seasons (season_name, status) "
                "VALUES (?, 'off')", (name,))
        # seed admin
        count = conn.execute("SELECT COUNT(*) AS c FROM users").fetchone()["c"]
        if count == 0:
            conn.execute(
                "INSERT INTO users (username, password, role) VALUES (?, ?, ?)",
                ("admin", hash_password("admin123"), ROLE_ADMIN))
        # seed settings
        for key, value in DEFAULT_SETTINGS.items():
            conn.execute(
                "INSERT OR IGNORE INTO settings (key, value) VALUES (?, ?)",
                (key, value))
        old = conn.execute(
            "SELECT value FROM settings WHERE key = 'currency_symbol'"
        ).fetchone()
        if old is not None and old["value"] == "$":
            conn.execute(
                "UPDATE settings SET value = ? WHERE key = 'currency_symbol'",
                ("ج.م",))
        # backfill invoice_number for legacy rows
        rows = conn.execute(
            "SELECT id FROM invoices WHERE invoice_number IS NULL "
            "ORDER BY id").fetchall()
        if rows:
            max_row = conn.execute(
                "SELECT COALESCE(MAX(invoice_number), ?) AS m FROM invoices",
                (INVOICE_START_NUMBER,)).fetchone()
            nxt = max(int(max_row["m"]), INVOICE_START_NUMBER)
            for r in rows:
                nxt += 1
                conn.execute(
                    "UPDATE invoices SET invoice_number = ? WHERE id = ?",
                    (nxt, r["id"]))


# --- Users ---

def create_user(username, password, role=ROLE_USER):
    username = (username or "").strip()
    if not username:
        raise ValueError("اسم المستخدم مطلوب.")
    if not password or len(password) < 4:
        raise ValueError("كلمة المرور يجب أن تكون 4 أحرف على الأقل.")
    if role not in VALID_ROLES:
        raise ValueError(f"الدور يجب أن يكون أحد: {VALID_ROLES}.")
    with get_connection() as conn:
        try:
            cur = conn.execute(
                "INSERT INTO users (username, password, role) VALUES (?, ?, ?)",
                (username, hash_password(password), role))
        except sqlite3.IntegrityError:
            raise ValueError(f"اسم المستخدم '{username}' موجود بالفعل.")
        return int(cur.lastrowid)


def authenticate(username, password):
    with get_connection() as conn:
        row = conn.execute("SELECT * FROM users WHERE username = ?",
                           ((username or "").strip(),)).fetchone()
    if row is None or not verify_password(password, row["password"]):
        return None
    return {"id": row["id"], "username": row["username"],
            "role": row["role"]}


def get_all_users():
    with get_connection() as conn:
        rows = conn.execute(
            "SELECT id, username, role, created_at FROM users "
            "ORDER BY username COLLATE NOCASE").fetchall()
    return _rows_to_dicts(rows)


def update_user_password(user_id, new_password):
    if not new_password or len(new_password) < 4:
        raise ValueError("كلمة المرور يجب أن تكون 4 أحرف على الأقل.")
    with get_connection() as conn:
        cur = conn.execute("UPDATE users SET password = ? WHERE id = ?",
                           (hash_password(new_password), user_id))
        return cur.rowcount > 0


def update_user_role(user_id, role):
    if role not in VALID_ROLES:
        raise ValueError(f"الدور يجب أن يكون أحد: {VALID_ROLES}.")
    with get_connection() as conn:
        user = conn.execute("SELECT role FROM users WHERE id = ?",
                            (user_id,)).fetchone()
        if user is None:
            return False
        if user["role"] == ROLE_ADMIN and role != ROLE_ADMIN:
            admins = conn.execute(
                "SELECT COUNT(*) AS c FROM users WHERE role = ?",
                (ROLE_ADMIN,)).fetchone()["c"]
            if admins <= 1:
                raise ValueError("لا يمكن تخفيض آخر مسؤول متبقٍ.")
        cur = conn.execute("UPDATE users SET role = ? WHERE id = ?",
                           (role, user_id))
        return cur.rowcount > 0


def delete_user(user_id):
    with get_connection() as conn:
        user = conn.execute("SELECT role FROM users WHERE id = ?",
                            (user_id,)).fetchone()
        if user is None:
            return False
        if user["role"] == ROLE_ADMIN:
            admins = conn.execute(
                "SELECT COUNT(*) AS c FROM users WHERE role = ?",
                (ROLE_ADMIN,)).fetchone()["c"]
            if admins <= 1:
                raise ValueError("لا يمكن حذف آخر مسؤول متبقٍ.")
        cur = conn.execute("DELETE FROM users WHERE id = ?", (user_id,))
        return cur.rowcount > 0

def add_product(code, name, supplier=""):
    code = (code or "").strip()
    name = (name or "").strip()
    if not code:
        raise ValueError("كود المنتج مطلوب.")
    if not name:
        raise ValueError("اسم المنتج مطلوب.")
    with get_connection() as conn:
        try:
            cur = conn.execute(
                "INSERT INTO products (code, name, supplier) "
                "VALUES (?, ?, ?)",
                (code, name, (supplier or "").strip()))
        except sqlite3.IntegrityError:

def _next_invoice_number(conn) -> int:
    row = conn.execute(
        "SELECT COALESCE(MAX(invoice_number), ?) AS m FROM invoices",
        (INVOICE_START_NUMBER,)).fetchone()
    return max(int(row["m"]), INVOICE_START_NUMBER) + 1


def get_next_invoice_number() -> int:
    with get_connection() as conn:
        return _next_invoice_number(conn)


def create_invoice(customer_name="", customer_phone="", items=None,
                   discount=0.0, cashier_name="", created_at=None):
    items = list(items or [])
    if not items:
        raise ValueError("لا يمكن إنشاء فاتورة بدون أصناف.")
    discount = _money(discount)
    if discount < 0:
        raise ValueError("الخصم لا يمكن أن يكون سالباً.")
    created_at = created_at or _now()
    with get_connection() as conn:
        lines, demand, cache = [], {}, {}
        for raw in items:
            code = str(raw.get("product_code") or raw.get("code")
                       or "").strip()
            if not code:
                raise ValueError("كل سطر يجب أن يحتوي على كود منتج.")
            try:
                qty = int(raw.get("quantity", raw.get("qty", 0)))
            except (TypeError, ValueError):
                raise ValueError(f"كمية غير صالحة للمنتج '{code}'.")
            if qty <= 0:
                raise ValueError(
                    f"الكمية للمنتج '{code}' يجب أن تكون أكبر من صفر.")
            product = cache.get(code)
            if product is None:
                product = conn.execute(
                    "SELECT * FROM products WHERE code = ?",
                    (code,)).fetchone()
                if product is None:
                    raise ValueError(
                        f"المنتج بالكود '{code}' غير موجود.")
                cache[code] = product
            raw_price = raw.get("price")
            price = _money(product["wholesale_price"] if raw_price is None
                           else raw_price)
            if price < 0:
                raise ValueError(f"سعر غير صالح للمنتج '{code}'.")
            demand[code] = demand.get(code, 0) + qty
            lines.append({"product_code": code,
                          "product_name": product["name"],
                          "quantity": qty, "price": price,
                          "total_price": _money(price * qty)})
        for code, qty in demand.items():
            if int(cache[code]["stock_qty"]) < qty:
                raise ValueError(
                    f"المخزون غير كافٍ للمنتج '{cache[code]['name']}' "
                    f"(المتاح: {cache[code]['stock_qty']}, "
                    f"المطلوب: {qty}).")
        subtotal = _money(sum(l["total_price"] for l in lines))
        total_items = int(sum(l["quantity"] for l in lines))
        if discount > subtotal:
            raise ValueError("الخصم لا يمكن أن يتجاوز المجموع الفرعي.")
        net_total = _money(subtotal - discount)
        inv_num = _next_invoice_number(conn)
        cur = conn.execute(
            """INSERT INTO invoices
                   (invoice_number, customer_name, customer_phone,
                    total_items, subtotal, discount, net_total,
                    created_at, cashier_name)
               VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)""",
            (inv_num, (customer_name or "").strip(),
             (customer_phone or "").strip(),
             total_items, subtotal, discount, net_total, created_at,
             (cashier_name or "").strip()))
        invoice_id = int(cur.lastrowid)
        for line in lines:
            conn.execute(
                """INSERT INTO invoice_items
                       (invoice_id, product_code, product_name, quantity,
                        price, total_price) VALUES (?, ?, ?, ?, ?, ?)""",
                (invoice_id, line["product_code"], line["product_name"],
                 line["quantity"], line["price"], line["total_price"]))
            conn.execute(
                "UPDATE products SET stock_qty = stock_qty - ? "
                "WHERE code = ?",
                (line["quantity"], line["product_code"]))
    return get_invoice(invoice_id)

            if season["status"] == "on":

                clear_operational_data()
                conn.execute(
                    "UPDATE seasons SET status = 'off', closed_at = ? "
                    "WHERE id = ?", (_now(), season_id))


def backup_database(destination_path=None):
    os.makedirs(BACKUP_DIR, exist_ok=True)
    if destination_path is None:
        ts = datetime.now().strftime("%Y%m%d_%H%M%S")
        destination_path = os.path.join(
            BACKUP_DIR, f"sierra_backup_{ts}.db")
    destination_path = os.path.abspath(destination_path)
    os.makedirs(os.path.dirname(destination_path) or ".", exist_ok=True)
    with get_connection() as source:
        target = sqlite3.connect(destination_path)
        try:
            source.backup(target)
        finally:
            target.close()
    return destination_path


def restore_database_from(backup_path):
    backup_path = os.path.abspath(backup_path)
    if not os.path.isfile(backup_path):
        raise FileNotFoundError(
            f"ملف النسخة الاحتياطية غير موجود: {backup_path}")
    with open(backup_path, "rb") as f:
        header = f.read(16)
    if header != b"SQLite format 3\x00":
        raise ValueError(
            "الملف المحدد ليس قاعدة بيانات SQLite صالحة (.db).")
    try:
        with get_connection() as conn:
            conn.execute("PRAGMA wal_checkpoint(TRUNCATE);")
    except Exception:
        pass
    for suffix in ("-wal", "-shm"):
        sidecar = DB_PATH + suffix
        try:
            if os.path.exists(sidecar):
                os.remove(sidecar)
        except Exception:
            pass
    os.makedirs(os.path.dirname(DB_PATH) or ".", exist_ok=True)
    shutil.copy2(backup_path, DB_PATH)
    return DB_PATH

def get_sales_summary(date_from=None, date_to=None):
    clauses, params = _date_filter("created_at", date_from, date_to)
    query = """SELECT COUNT(*) AS invoice_count,
                      COALESCE(SUM(total_items), 0) AS total_items,
                      COALESCE(SUM(subtotal), 0) AS subtotal,
                      COALESCE(SUM(discount), 0) AS discount,
                      COALESCE(SUM(net_total), 0) AS net_total
               FROM invoices"""
    if clauses:
        query += " WHERE " + " AND ".join(clauses)
    with get_connection() as conn:
        row = conn.execute(query, params).fetchone()
    return {"invoice_count": int(row["invoice_count"]),
            "total_items": int(row["total_items"]),
            "subtotal": _money(row["subtotal"]),
            "discount": _money(row["discount"]),
            "net_total": _money(row["net_total"])}


def get_cost_of_goods_sold(date_from=None, date_to=None):
    clauses, params = _date_filter("i.created_at", date_from, date_to)
    query = """SELECT COALESCE(SUM(ii.quantity * COALESCE(p.cost_price, 0)), 0) AS cogs
               FROM invoice_items ii
               JOIN invoices i ON i.id = ii.invoice_id
               LEFT JOIN products p ON p.code = ii.product_code"""
    if clauses:
        query += " WHERE " + " AND ".join(clauses)
    with get_connection() as conn:
        return _money(conn.execute(query, params).fetchone()["cogs"])


def get_profit_report(date_from=None, date_to=None):
    sales = get_sales_summary(date_from, date_to)
    cogs = get_cost_of_goods_sold(date_from, date_to)
    expenses = get_total_expenses(date_from, date_to)
    revenue = _money(sales["net_total"])
    gross = _money(revenue - cogs)
    net = _money(gross - expenses)
    return {"revenue": revenue, "cogs": cogs, "gross_profit": gross,
            "expenses": expenses, "net_profit": net,
            "invoice_count": sales["invoice_count"],
            "total_items": sales["total_items"]}


def get_dashboard_stats():
    with get_connection() as conn:
        def scalar(q, p=()):
            r = conn.execute(q, p).fetchone()
            return r[0] if r is not None else 0
        return {

def get_setting(key, default=None):
    with get_connection() as conn:
        row = conn.execute("SELECT value FROM settings WHERE key = ?",
                           (key,)).fetchone()
    return default if row is None else row["value"]


def set_settings(values):
    with get_connection() as conn:
        for key, value in values.items():
            conn.execute(
                "INSERT INTO settings (key, value) VALUES (?, ?) "
                "ON CONFLICT(key) DO UPDATE SET value = excluded.value",
                (key, "" if value is None else str(value)))


def get_all_settings():
    with get_connection() as conn:
        rows = conn.execute("SELECT key, value FROM settings").fetchall()
    settings = dict(DEFAULT_SETTINGS)
    settings.update({r["key"]: r["value"] for r in rows})
    return settings

    return get_setting("currency_symbol",
                       DEFAULT_SETTINGS["currency_symbol"]) or "ج.م"

                       DEFAULT_SETTINGS["invoice_header"]) or ""

def money_str(value, currency=None):
    if currency is None:
        currency = get_currency_symbol()
    try:
        return f"{float(value):,.2f} {currency}"
    except (TypeError, ValueError):
        return f"0.00 {currency}"


# --- Receipt formatter ---

def _receipt_width(paper_mm):
    return 32 if "58" in str(paper_mm) else 42


def format_receipt(invoice, header, footer, currency="ج.م",
                   paper_mm="80mm"):
    W = _receipt_width(paper_mm)

    def center(s):
        return s.center(W)[:W]

    def sep(ch="="):
        return ch * W

    def row(left, right):
        gap = W - len(left) - len(right)
        return ((left + " " + right)[:W] if gap < 1
                else left + " " * gap + right)

    out = [sep("=")]
    out.append(center("SIERRA STORE"))
    for h in (header or "").splitlines():
        if h.strip() and h.strip() != "SIERRA STORE":
            out.append(center(h))
    out.append(sep("="))
    out.append(f"Invoice No: {invoice.get('invoice_number', invoice['id'])}")
    out.append(f"Date      : {invoice['created_at']}")
    if invoice.get("cashier_name"):
        out.append(f"Cashier   : {invoice['cashier_name']}")
    if invoice.get("customer_name"):
        out.append(f"Customer  : {invoice['customer_name']}")
    if invoice.get("customer_phone"):
        out.append(f"Phone     : {invoice['customer_phone']}")
    out.append(sep("-"))
    out.append(row("Item / Qty / Price / Total", ""))
    out.append(sep("-"))
    for it in invoice.get("items", []):
        name = (it["product_name"] or "")[:max(1, W - 20)]
        line = f"{name} x{it['quantity']} @ {currency}{it['price']:.2f}"
        out.append(row(line[:W - 12], f"{currency}{it['total_price']:.2f}"))
    out.append(sep("-"))
    out.append(row("توتال القطع", str(invoice["total_items"])))
    out.append(row("توتال الفلوس",
                   f"{currency}{invoice['net_total']:.2f}"))
    if invoice["discount"] > 0:
        out.append(row("Discount",
                       f"-{currency}{invoice['discount']:.2f}"))
        out.append(row("Subtotal",
                       f"{currency}{invoice['subtotal']:.2f}"))
    out.append(sep("="))
    for f in (footer or "").splitlines():
        out.append(center(f))
    return "\n".join(out)

def try_send_to_printer(receipt_text: str) -> Optional[str]:
    """Try to print via the configured thermal printer.

    Returns a status message on success, or None if printing is not available
    on this platform. The actual printing is optional; the receipt is always
    shown on screen as a fallback."""
    printer_name = get_printer_name()
    if not printer_name:
        return None
    try:
        if os.name == "nt":
            # On Windows, write raw text to the printer via win32print
            import win32print  # type: ignore
            handle = win32print.OpenPrinter(printer_name)
            try:
                job = win32print.StartDocPrinter(
                    handle, 1, ("SIERRA STORE", None, "RAW"))
                try:
                    win32print.StartPagePrinter(handle)
                    win32print.WritePrinter(
                        handle, receipt_text.encode("utf-8"))
                    win32print.EndPagePrinter(handle)
                finally:
                    win32print.EndDocPrinter(handle)
            finally:
                win32print.ClosePrinter(handle)
            return f"تم الإرسال إلى الطابعة: {printer_name}"
    except Exception:
        pass
    return None


def show_receipt_dialog(page, invoice, currency, paper_mm="80mm"):
    receipt = format_receipt(invoice, get_invoice_header(),
                             get_invoice_footer(), currency, paper_mm)

    field = ft.TextField(
        value=receipt, multiline=True, min_lines=20, max_lines=28,
        read_only=True,
        text_style=ft.TextStyle(font_family="Courier New",
                                size=FS_SMALL, color=TEXT_PRIMARY),
        bgcolor=BG_INPUT, border_color=BORDER,
        border_radius=RADIUS_SM, width=480)

    def copy(e):
        try:
            page.set_clipboard(receipt)
            snack(page, "تم نسخ الإيصال إلى الحافظة.", kind="success")
        except Exception:
            snack(page, "النسخ غير مدعوم.", kind="error")

    def send_print(e):
        msg = try_send_to_printer(receipt)
        if msg:
            snack(page, msg, kind="success")
        else:
            snack(page,
                  "لم يتم العثور على طابعة حرارية مُهيأة "
                  "(يمكن ضبط اسم الطابعة من الإعدادات).",
                  kind="warning")

    def close(e):
        page.close(dlg)

    dlg = ft.AlertDialog(
        modal=True, bgcolor=BG_CARD,
        title=ft.Text(
            f"إيصال — فاتورة رقم "
            f"{invoice.get('invoice_number', invoice['id'])}",
            color=TEXT_PRIMARY, weight=ft.FontWeight.W_700,
            size=FS_SECTION),
        content=field,
        actions=[ghost_button("نسخ", copy, icon=ft.icons.COPY, width=110),
                 ghost_button("طباعة", send_print,
                              icon=ft.icons.PRINT, width=110),
                 primary_button("تم", close, width=110)])
    page.open(dlg)

                           spacing=0, tight=True, expand=True),
                 ft.Text(money_str(p["wholesale_price"], self.currency),
                         color=TEXT_PRIMARY, size=FS_BODY,
                         weight=ft.FontWeight.W_700),
                 ft.Container(width=SP_SM),
                 chip(f"متوفر {p['stock_qty']}", color=stock_color,
                      bg=BG_CARD)],
                vertical_alignment=ft.CrossAxisAlignment.CENTER),
            padding=ft.padding.symmetric(horizontal=12, vertical=6),
            border_radius=RADIUS_SM, on_click=_click, ink=True,
            on_hover=_hover_ghost)

    def _add_to_cart(self, product_id):
        product = get_product_by_id(product_id)
        if product is None:
            return
        if product["stock_qty"] <= 0:
            snack(self.page,
                  f"'{product['name']}' غير متوفر في المخزون.",
                  kind="error")
            return
        for line in self.cart:
            if line["product_code"] == product["code"]:
                if line["quantity"] + 1 > product["stock_qty"]:
                    snack(self.page,
                          f"المتوفر فقط {product['stock_qty']} من "
                          f"'{product['name']}'.", kind="warning")
                    return
                line["quantity"] += 1
                self._refresh_cart()
                return
        self.cart.append({"product_code": product["code"],
                          "product_name": product["name"], "quantity": 1,
                          "price": float(product["wholesale_price"]),
                          "stock": int(product["stock_qty"])})
        self._refresh_cart()

    def _inc(self, idx):
        line = self.cart[idx]
        if line["quantity"] + 1 > line["stock"]:
            snack(self.page, f"المتوفر فقط {line['stock']}.",
                  kind="warning")
            return
        line["quantity"] += 1
        self._refresh_cart()

    def _dec(self, idx):
        self.cart[idx]["quantity"] -= 1
        if self.cart[idx]["quantity"] <= 0:
            self.cart.pop(idx)
        self._refresh_cart()

    def _remove(self, idx):
        self.cart.pop(idx)
        self._refresh_cart()

    def _refresh_cart(self):
        self.cart_area.controls.clear()
        if not self.cart:
            self.cart_area.controls.append(self._empty_cart())
        else:
            for idx, line in enumerate(self.cart):
                self.cart_area.controls.append(self._cart_row(idx, line))
        self._recalc_totals()
        try:
            self.cart_area.update()
        except Exception:
            pass

    def _cart_row(self, idx, line):
        subtotal = line["price"] * line["quantity"]
        qty_controls = ft.Row(
            [icon_button(ft.icons.REMOVE, lambda e, i=idx: self._dec(i),
                         tooltip="تقليل", size=28),
             ft.Container(content=ft.Text(str(line["quantity"]),
                                          color=TEXT_PRIMARY,
                                          size=FS_BODY,
                                          weight=ft.FontWeight.W_700),
                          width=32, alignment=ft.alignment.center),
             icon_button(ft.icons.ADD, lambda e, i=idx: self._inc(i),
                         tooltip="زيادة", size=28)],
            spacing=2, vertical_alignment=ft.CrossAxisAlignment.CENTER)
        return ft.Container(
            content=ft.Row(
                [ft.Container(content=ft.Column(
                    [ft.Text(line["product_name"], color=TEXT_PRIMARY,
                             size=FS_BODY,
                             weight=ft.FontWeight.W_600),
                     ft.Text(line["product_code"], color=TEXT_MUTED,
                             size=FS_TINY)],
                    spacing=0, tight=True), expand=4),
                 ft.Container(qty_controls, width=140),
                 ft.Container(content=ft.Text(
                     money_str(line["price"], self.currency),
                     color=TEXT_SECONDARY, size=FS_SMALL), width=110),
                 ft.Container(content=ft.Text(
                     money_str(subtotal, self.currency),
                     color=TEXT_PRIMARY, size=FS_BODY,
                     weight=ft.FontWeight.W_700), width=110),
                 icon_button(ft.icons.CLOSE,
                             lambda e, i=idx: self._remove(i),
                             tooltip="إزالة", color=DANGER, size=30)],
                spacing=SP_SM,
                vertical_alignment=ft.CrossAxisAlignment.CENTER),
            padding=ft.padding.symmetric(horizontal=SP_MD, vertical=6),
            border=ft.border.only(bottom=ft.BorderSide(1, BORDER)))

    def _on_discount_toggle(self, e):
        self.discount_enabled = bool(self.discount_check.value)
        self.discount_container.visible = self.discount_enabled
        if not self.discount_enabled:
            self.discount_value = 0.0
            self.discount_input.value = "0"
        try:
            self.discount_container.update()
        except Exception:
            pass
        self._recalc_totals()

    def _on_discount_type_change(self, e):
        self.discount_type = self.discount_type_dd.value or "ثابت"
        self._recalc_totals()

    def _on_discount_value_change(self, e):
        try:
            self.discount_value = float(self.discount_input.value or 0)
        except ValueError:
            self.discount_value = 0.0
        self._recalc_totals()

    def _recalc_totals(self):
        total_items = sum(l["quantity"] for l in self.cart)
        subtotal = sum(l["price"] * l["quantity"] for l in self.cart)
        discount = 0.0
        if self.discount_enabled:
            if self.discount_type == "نسبة مئوية":
                discount = subtotal * (self.discount_value / 100.0)
            else:
                discount = self.discount_value
            discount = max(0.0, min(discount, subtotal))
        net = subtotal - discount
        try:
            self.items_count_text.value = str(total_items)
            self.subtotal_text.value = money_str(subtotal, self.currency)
            self.discount_text.value = (
                f"- {money_str(discount, self.currency)}")
            self.net_total_text.value = money_str(net, self.currency)
            self.items_count_text.update()
            self.subtotal_text.update()
            self.discount_text.update()
            self.net_total_text.update()
        except Exception:
            pass

    def _on_save_request(self, e):
        if not self.cart:
            snack(self.page, "السلة فارغة.", kind="warning")
            return
        subtotal = sum(l["price"] * l["quantity"] for l in self.cart)
        net = subtotal
        if self.discount_enabled:
            d = (subtotal * (self.discount_value / 100.0)
                 if self.discount_type == "نسبة مئوية"
                 else self.discount_value)
            d = max(0.0, min(d, subtotal))
            net = subtotal - d
        msg = (f"هل تريد حفظ الفاتورة؟\n"
               f"عدد القطع: {sum(l['quantity'] for l in self.cart)}\n"
               f"الإجمالي: {money_str(net, self.currency)}")
        confirm_dialog(
            self.page, "تأكيد الحفظ", msg,
            lambda: self._perform_save(), confirm_text="تأكيد",
            cancel_text="إلغاء", danger=False)

    def _perform_save(self):
        subtotal = sum(l["price"] * l["quantity"] for l in self.cart)
        discount = 0.0
        if self.discount_enabled:
            if self.discount_type == "نسبة مئوية":
                discount = subtotal * (self.discount_value / 100.0)
            else:
                discount = self.discount_value
            discount = max(0.0, min(discount, subtotal))
        try:
            invoice = create_invoice(
                customer_name=self.customer_name.value or "",
                customer_phone=self.customer_phone.value or "",
                items=[{"product_code": l["product_code"],
                        "quantity": l["quantity"],
                        "price": l["price"]} for l in self.cart],
                discount=discount, cashier_name=self.user["username"])
        except Exception as ex:
            snack(self.page, str(ex), kind="error")
            return
        self.cart.clear()
        self.customer_name.value = ""
        self.customer_phone.value = ""
        self.discount_check.value = False
        self.discount_enabled = False
        self.discount_container.visible = False
        self.discount_input.value = "0"
        self.discount_value = 0.0
        self._refresh_cart()
        self._update_header()
        try:
            self.customer_name.update()
            self.customer_phone.update()
            self.discount_check.update()
            self.discount_container.update()
        except Exception:
            pass
        snack(self.page,
              f"تم حفظ الفاتورة رقم {invoice['invoice_number']}.",
              kind="success")
        show_receipt_dialog(self.page, invoice, self.currency,
                            self.paper_mm)

    def _on_clear_cart(self, e):
        if not self.cart:
            return
        self.cart.clear()
        self._refresh_cart()

class SalesView:

    def _can_modify(self, inv: dict) -> bool:
        if self.user["role"] == ROLE_ADMIN:
            return True
        return (inv.get("cashier_name") or "") == self.user["username"]

        header = ft.Row(
            [page_title("المبيعات", "الفواتير المسجلة"),

            vertical_alignment=ft.CrossAxisAlignment.CENTER,

            [header, ft.Container(height=SP_SM),
             card(ft.Column([self.table_container], expand=True, spacing=0),
                  padding=0, hover=False, expand=True)],
            spacing=0, expand=True)

    def _on_search(self, e):
        self._refresh()

        invoices = get_all_invoices(search=self.search.value or "")
        rows, ctx = [], []
        for inv in invoices:
            can_edit = self._can_modify(inv)

                [icon_button(ft.icons.PRINT,
                             lambda e, inv=inv: self._reprint(inv),

                             lambda e, inv=inv: self._confirm_delete(inv),

             ("أنشئت بواسطة", 160), ("", 140)],

        try:
            self.table_container.update()
        except Exception:
            pass

    def _reprint(self, inv):
        full = get_invoice(inv["id"])
        if full is None:
            snack(self.page, "الفاتورة غير موجودة.", kind="error")
            return
        s = get_all_settings()
        show_receipt_dialog(self.page, full, self.currency,
                            s.get("printer_paper_width", "80mm"))

    def _open_edit_dialog(self, inv):
        if not self._can_modify(inv):
            snack(self.page,
                  "لا تملك صلاحية تعديل هذه الفاتورة.",
                  kind="error")
            return
        name = text_field("اسم العميل",
                          value=inv.get("customer_name") or "",
                          width=300)
        phone = text_field("هاتف العميل",
                           value=inv.get("customer_phone") or "",
                           width=300)
        discount = text_field(
            "الخصم", value=f"{inv.get('discount', 0):.2f}", width=200,
            keyboard_type=ft.KeyboardType.NUMBER)
        error = ft.Text("", color=DANGER, size=FS_SMALL, visible=False)

        def save(e):
            error.visible = False
            try:
                update_invoice(inv["id"], customer_name=name.value,
                               customer_phone=phone.value,
                               discount=discount.value)
            except Exception as ex:
                error.value = str(ex)
                error.visible = True
                error.update()
                return
            self.page.close(dlg)
            self._refresh()
            snack(self.page, "تم تحديث الفاتورة.", kind="success")

        def cancel(e=None):
            self.page.close(dlg)

        content = ft.Column(
            [name, phone, discount, error],
            spacing=SP_SM, width=340)
        dlg = ft.AlertDialog(
            modal=True, bgcolor=BG_CARD,
            title=ft.Text(
                f"تعديل الفاتورة #{inv.get('invoice_number', inv['id'])}",
                color=TEXT_PRIMARY, weight=ft.FontWeight.W_700,
                size=FS_SECTION),
            content=content,
            actions=[ghost_button("إلغاء", cancel, width=110),
                     primary_button("حفظ", save, icon=ft.icons.CHECK,
                                    width=110)])
        self.page.open(dlg)

    def _confirm_delete(self, inv):
        if not self._can_modify(inv):
            snack(self.page,
                  "لا تملك صلاحية حذف هذه الفاتورة.",
                  kind="error")
            return

        def do_delete():
            try:
                delete_invoice(inv["id"])
                self._refresh()
                snack(self.page, "تم حذف الفاتورة وإرجاع المخزون.",
                      kind="success")
            except Exception as ex:
                snack(self.page, str(ex), kind="error")

            self.page, "حذف الفاتورة",
            f"حذف الفاتورة #{inv.get('invoice_number', inv['id'])}؟ "
            "سيتم إرجاع الكميات إلى المخزون.",

class PurchasesView:

        self.products = get_all_products()
        self.product_dd = keyed_dropdown(
            "المنتج",
            [(p["code"], p["name"]) for p in self.products],

        self.name_field = text_field("اسم المنتج", read_only=True,
                                     width=200)
        self.supplier_field = text_field("المورد", read_only=True,
                                         width=180)

            "الكمية الواردة", width=130,
            keyboard_type=ft.KeyboardType.NUMBER)
        self.cost_field = text_field(
            "سعر التكلفة", width=130,
            keyboard_type=ft.KeyboardType.NUMBER)
        self.selling_field = text_field(
            "سعر البيع", width=130,
            keyboard_type=ft.KeyboardType.NUMBER)
        self.table_container = ft.Container(expand=True)

    def _on_product_change(self, e):
        code = self.product_dd.value
        if code:
            p = get_product_by_code(code)
            if p:
                self.name_field.value = p["name"]
                self.supplier_field.value = p.get("supplier") or ""
                self.selling_field.value = (
                    f"{p['wholesale_price']:.2f}"
                    if p["wholesale_price"] else "")
                self.cost_field.value = (
                    f"{p['cost_price']:.2f}"
                    if p["cost_price"] else "")
        else:
            self.name_field.value = ""
            self.supplier_field.value = ""
            self.selling_field.value = ""
            self.cost_field.value = ""
        try:
            self.name_field.update()
            self.supplier_field.update()
            self.selling_field.update()
            self.cost_field.update()
        except Exception:
            pass

    def build(self):
        save_btn = primary_button("حفظ", self._on_add,
                                  icon=ft.icons.SAVE, width=120)
        form = card(
            ft.Column(
                [section_title("تسجيل عملية شراء جديدة"),
                 ft.Container(height=SP_SM),
                 ft.Row(
                     [self.product_dd, self.name_field,
                      self.supplier_field, self.qty_field,
                      self.cost_field, self.selling_field, save_btn],
                     spacing=SP_MD,
                     vertical_alignment=ft.CrossAxisAlignment.CENTER,
                     wrap=True)],
                spacing=0),
            padding=SP_MD, hover=False)

        self._refresh_table()

        return ft.Column(
            [page_title("المشتريات", "تسجيل المشتريات من الموردين"),
             ft.Container(height=SP_SM),
             form,
             ft.Container(height=SP_SM),
             card(ft.Column([self.table_container],
                            expand=True, spacing=0),
                  padding=0, hover=False, expand=True)],
            spacing=0, expand=True)

    def _on_add(self, e):
        code = self.product_dd.value
        if not code:
            snack(self.page, "اختر المنتج أولاً.", kind="warning")
            return
        try:
            create_purchase(
                product_code=code,
                quantity=self.qty_field.value,
                purchase_price=self.cost_field.value,
                selling_price=self.selling_field.value)
        except Exception as ex:
            snack(self.page, str(ex), kind="error")
            return
        self.product_dd.value = None
        self.name_field.value = ""
        self.supplier_field.value = ""
        self.qty_field.value = ""
        self.cost_field.value = ""
        self.selling_field.value = ""
        try:
            self.product_dd.update()
            self.name_field.update()
            self.supplier_field.update()
            self.qty_field.update()
            self.cost_field.update()
            self.selling_field.update()
        except Exception:

class SeasonsView:
    def __init__(self, app, page, user):
        self.app = app
        self.page = page
        self.user = user
        self.container = ft.Container(expand=True)

    def build(self):
        self._refresh()
        return self.container

    def _refresh(self):
        seasons = get_all_seasons()
        cards = ft.Row(spacing=SP_LG, wrap=True)
        for s in seasons:
            is_on = s["status"] == "on"

            def make_toggle(sid=s["id"]):
                def _toggle(e, sid=sid):
                    new_val = bool(e.control.value)
                    current = next(
                        (x for x in get_all_seasons() if x["id"] == sid),
                        None)
                    if current is None:
                        return
                    if new_val == (current["status"] == "on"):
                        return

                    def commit():
                        try:
                            set_season_status(sid, new_val)
                            self._refresh()
                            snack(
                                self.page,
                                ("تم تفعيل الموسم." if new_val
                                 else "تم إيقاف الموسم وحفظ نسخة "
                                      "احتياطية ثم مسح البيانات."),
                                kind="success")
                        except Exception as ex:
                            snack(self.page, str(ex), kind="error")

                    if new_val:
                        commit()
                    else:
                        confirm_dialog(
                            self.page, "إيقاف الموسم",
                            "سيتم أخذ نسخة احتياطية تلقائية ثم مسح جميع "
                            "حركات الموسم (المخزون، المشتريات، المبيعات، "
                            "المصروفات). الموردون وأرقام الفواتير تبقى "
                            "محفوظة. متابعة؟",
                            commit, confirm_text="إيقاف", danger=True)
                return _toggle

            sw = ft.Switch(value=is_on, on_change=make_toggle(),
                           active_color=SUCCESS)
            status_color = SUCCESS if is_on else TEXT_MUTED
            status_text = "مفعّل" if is_on else "متوقف"

            cards.controls.append(ft.Container(
                content=ft.Column(
                    [ft.Row(
                        [ft.Icon(ft.icons.CALENDAR_MONTH, size=24,
                                 color=status_color),
                         ft.Container(width=SP_SM),
                         ft.Text(s["season_name"], color=TEXT_PRIMARY,
                                 size=FS_SECTION,
                                 weight=ft.FontWeight.W_800,
                                 expand=True)],
                        vertical_alignment=ft.CrossAxisAlignment.CENTER),
                     ft.Container(height=SP_SM),
                     ft.Row(
                         [ft.Text("الحالة:", color=TEXT_MUTED,
                                  size=FS_SMALL),
                          chip(status_text, color=status_color,
                               bg=BG_INPUT)],
                         spacing=SP_SM),
                     ft.Container(height=SP_SM),
                     ft.Row(
                         [ft.Text("التشغيل/الإيقاف",
                                  color=TEXT_SECONDARY, size=FS_SMALL),
                          ft.Container(expand=True), sw],
                         vertical_alignment=ft.CrossAxisAlignment.CENTER),
                     ft.Container(height=SP_SM), divider(),
                     ft.Container(height=SP_SM),
                     ft.Text(
                         f"تاريخ التفعيل: {s['activated_at'] or '—'}",
                         color=TEXT_MUTED, size=FS_TINY),
                     ft.Text(
                         f"تاريخ الإغلاق: {s['closed_at'] or '—'}",
                         color=TEXT_MUTED, size=FS_TINY)],
                    spacing=4),
                bgcolor=BG_CARD, border=ft.border.all(1, BORDER),
                border_radius=RADIUS_MD, padding=SP_MD,
                width=320, height=260, on_hover=_hover_card))

        history_rows = []
        for s in seasons:
            status_color = SUCCESS if s["status"] == "on" else TEXT_MUTED
            history_rows.append([

                     color=status_color, bg=BG_INPUT),
                ft.Text(s["activated_at"] or "—", color=TEXT_SECONDARY,
                        size=FS_SMALL),
                ft.Text(s["closed_at"] or "—", color=TEXT_SECONDARY,
                        size=FS_SMALL),
            ])

        self.container.content = ft.Column(
            [page_title("المواسم",
                        "تشغيل/إيقاف المواسم الثلاثة المتعاقبة"),
             ft.Container(height=SP_MD),
             cards,
             ft.Container(height=SP_LG),
             card(ft.Column([section_title("سجل المواسم"),
                             ft.Container(height=SP_SM),
                             data_table(
                                 self.page,
                                 [("الموسم", 220), ("الحالة", 110),
                                  ("تاريخ التفعيل", 190),
                                  ("تاريخ الإغلاق", 190)],
                                 history_rows,
                                 "لا توجد مواسم.")],
                            expand=True, spacing=0),
                  padding=SP_MD, hover=False, expand=True)],
            spacing=0, expand=True)

        for key, label in tabs:
            is_active = self.report_type == key
            btn = ft.Container(
                content=ft.Text(
                    label,
                    color=BG_BLACK if is_active else TEXT_PRIMARY,
                    size=FS_SMALL, weight=ft.FontWeight.W_700),
                bgcolor=ACCENT if is_active else BG_INPUT,
                border_radius=RADIUS_SM,
                padding=ft.padding.symmetric(horizontal=14, vertical=8),
                on_click=lambda e, k=key: self._switch_tab(k),
                ink=True)
            if not is_active:
                btn.on_hover = _hover_ghost
            row.controls.append(btn)
        return row

    def _switch_tab(self, key):
        self.report_type = key
        self.tabs_container.content = self._build_tabs()
        self._refresh()
        try:
            self.tabs_container.update()
        except Exception:
            pass

    def _quick_range(self, days):
        today = date.today()
        self.date_from.value = (
            today - timedelta(days=days - 1)).isoformat()
        self.date_to.value = today.isoformat()
        try:
            self.date_from.update()
            self.date_to.update()
        except Exception:
            pass
        self._refresh()

    def _quick_range_month(self):
        today = date.today()
        self.date_from.value = today.replace(day=1).isoformat()
        self.date_to.value = today.isoformat()
        try:
            self.date_from.update()
            self.date_to.update()
        except Exception:
            pass
        self._refresh()

    def _refresh(self, e=None):
        dfrom = self.date_from.value or None
        dto = self.date_to.value or None
        try:
            report = get_profit_report(dfrom, dto)
        except Exception:
            report = {"revenue": 0, "cogs": 0, "gross_profit": 0,
                      "expenses": 0, "net_profit": 0,
                      "invoice_count": 0, "total_items": 0}

        self.summary_row.controls = [
            self._stat("إجمالي المبيعات",
                       money_str(report["revenue"], self.currency),
                       TEXT_PRIMARY, ft.icons.POINT_OF_SALE),
            self._stat("إجمالي القطع",
                       str(report["total_items"]),
                       TEXT_PRIMARY, ft.icons.INVENTORY_2),
            self._stat("المصروفات",
                       money_str(report["expenses"], self.currency),
                       WARNING, ft.icons.RECEIPT_LONG),
            self._stat("صافي الربح",

                       SUCCESS if report["net_profit"] >= 0 else DANGER,
                       ft.icons.SAVINGS),

            pass

                      ft.Text(value, color=color, size=18,
                              weight=ft.FontWeight.W_800)],
                     spacing=2, tight=True, expand=True)],
                vertical_alignment=ft.CrossAxisAlignment.CENTER),

    def _build_report_text(self):
        W = 60
        titles = {
            "sales": "تقرير المبيعات",
            "purchases": "تقرير المشتريات",
            "expenses": "تقرير المصروفات",
            "inventory": "تقرير المخزون",
            "invoice_range": "تقرير أرقام الفواتير",
        }
        lines = ["=" * W, "SIERRA STORE".center(W),
                 titles.get(self.report_type, "تقرير").center(W),
                 "-" * W,
                 f"من: {self.date_from.value or '—'}   "
                 f"إلى: {self.date_to.value or '—'}",
                 "=" * W]
        if self.current_headers:
            lines.append(" | ".join(self.current_headers))
            lines.append("-" * W)
        for row in self.current_rows_data:
            lines.append(" | ".join(str(x) for x in row))
        lines.append("=" * W)
        lines.append(f"عدد السجلات: {len(self.current_rows_data)}")
        return "\n".join(lines)

    def _on_print(self, e):
        text = self._build_report_text()

        def do_copy(e):
            try:
                self.page.set_clipboard(text)
                snack(self.page, "تم نسخ التقرير إلى الحافظة.",
                      kind="success")
            except Exception:
                snack(self.page, "النسخ غير مدعوم.", kind="error")

        def send_print(e):
            msg = try_send_to_printer(text)
            if msg:
                snack(self.page, msg, kind="success")
            else:
                snack(self.page,
                      "لم يتم العثور على طابعة حرارية مُهيأة "
                      "(يمكن ضبط اسم الطابعة من الإعدادات).",
                      kind="warning")

        def close(e):
            self.page.close(dlg)

        field = ft.TextField(
            value=text, multiline=True, min_lines=18, max_lines=28,
            read_only=True,
            text_style=ft.TextStyle(font_family="Courier New",
                                    size=FS_SMALL,
                                    color=TEXT_PRIMARY),
            bgcolor=BG_INPUT, border_color=BORDER,
            border_radius=RADIUS_SM, width=540)

        dlg = ft.AlertDialog(
            modal=True, bgcolor=BG_CARD,
            title=ft.Text("طباعة التقرير", color=TEXT_PRIMARY,
                          weight=ft.FontWeight.W_700,
                          size=FS_SECTION),
            content=field,
            actions=[ghost_button("نسخ", do_copy, icon=ft.icons.COPY,
                                  width=110),
                     ghost_button("طباعة", send_print,
                                  icon=ft.icons.PRINT, width=110),
                     primary_button("إغلاق", close, width=110)])
        self.page.open(dlg)

                (lambda uid=u["id"], un=u["username"]:
                 self._open_change_password(uid, un)),
                (lambda uid=u["id"], un=u["username"], s=is_self:
                 self._confirm_delete_user(uid, un, s))))
        self.users_container.content = data_table(
            self.page,
            [("اسم المستخدم", 220), ("الدور", 140),
             ("تاريخ الإنشاء", 190), ("", 120)],
            rows, "لا يوجد مستخدمون.", ctx)
        try:
            self.users_container.update()
        except Exception:
            pass

    def _open_change_password(self, user_id, username):
        new_pw = text_field("كلمة المرور الجديدة", password=True,
                            width=300)
        confirm = text_field("تأكيد كلمة المرور", password=True,
                             width=300)
        error = ft.Text("", color=DANGER, size=FS_SMALL, visible=False)

        def save(e):
            error.visible = False
            if new_pw.value != confirm.value:
                error.value = "كلمتا المرور غير متطابقتين."
                error.visible = True
                error.update()
                return
            try:
                update_user_password(user_id, new_pw.value)
            except Exception as ex:
                error.value = str(ex)
                error.visible = True
                error.update()
                return
            self.page.close(dlg)
            snack(self.page,
                  f"تم تحديث كلمة المرور لـ '{username}'.",
                  kind="success")

        def cancel(e=None):
            self.page.close(dlg)

        dlg = ft.AlertDialog(
            modal=True, bgcolor=BG_CARD,
            title=ft.Text(f"تغيير كلمة المرور — {username}",
                          color=TEXT_PRIMARY,
                          weight=ft.FontWeight.W_700,
                          size=FS_SECTION),
            content=ft.Column([new_pw, confirm, error],
                              spacing=SP_SM, width=320),
            actions=[ghost_button("إلغاء", cancel, width=110),
                     primary_button("تحديث", save, icon=ft.icons.SAVE,
                                    width=110)])
        self.page.open(dlg)

    def _toggle_role(self, user_id, current_role):
        new_role = (ROLE_USER if current_role == ROLE_ADMIN
                    else ROLE_ADMIN)
        try:
            update_user_role(user_id, new_role)
            self._refresh_users()
            snack(self.page, f"تم تغيير الدور إلى '{new_role}'.",
                  kind="success")
        except Exception as ex:
            snack(self.page, str(ex), kind="error")

    def _confirm_delete_user(self, user_id, username, is_self):
        if is_self:
            snack(self.page, "لا يمكنك حذف حسابك الحالي.",
                  kind="warning")
            return

        def do_delete():
            try:
                delete_user(user_id)
                self._refresh_users()
                snack(self.page, f"تم حذف المستخدم '{username}'.",
                      kind="success")
            except Exception as ex:
                snack(self.page, str(ex), kind="error")

        confirm_dialog(self.page, "حذف المستخدم",
                       f"حذف المستخدم '{username}'؟ "
                       "لا يمكن التراجع عن هذا الإجراء.", do_delete,
                       confirm_text="حذف", danger=True)

    def _add_picker(self, picker):
        try:
            if hasattr(self.page, "services"):
                self.page.services.append(picker)
            elif hasattr(self.page, "overlay"):
                self.page.overlay.append(picker)
            self.page.update()
        except Exception:
            pass

    def _remove_picker(self, picker):
        try:
            if hasattr(self.page, "services") \
                    and picker in self.page.services:
                self.page.services.remove(picker)
        except Exception:
            pass
        try:
            if hasattr(self.page, "overlay") \
                    and picker in self.page.overlay:
                self.page.overlay.remove(picker)
        except Exception:
            pass
        try:
            self.page.update()
        except Exception:
            pass

    @staticmethod
    def _extract_path(e):
        if e is None:
            return None
        path = getattr(e, "path", None)
        if path:
            return path
        files = getattr(e, "files", None)
        if files:
            try:
                p = getattr(files[0], "path", None)
                if p:
                    return p
            except Exception:
                return None
        return None

    async def _async_save(self, picker, default_name):
        try:
            await picker.save_file(
                dialog_title="حفظ النسخة الاحتياطية",
                file_name=default_name,
                allowed_extensions=["db"])
        except TypeError:
            try:
                picker.save_file(dialog_title="حفظ النسخة الاحتياطية",
                                 file_name=default_name,
                                 allowed_extensions=["db"])
            except Exception as ex:
                snack(self.page, str(ex), kind="error")
                self._remove_picker(picker)
        except Exception as ex:
            snack(self.page, str(ex), kind="error")
            self._remove_picker(picker)

    async def _async_pick(self, picker):
        try:
            await picker.pick_files(
                dialog_title="اختر ملف النسخة الاحتياطية (.db)",
                allow_multiple=False,
                allowed_extensions=["db"])
        except TypeError:
            try:
                picker.pick_files(
                    dialog_title="اختر ملف النسخة الاحتياطية (.db)",
                    allow_multiple=False,
                    allowed_extensions=["db"])
            except Exception as ex:
                snack(self.page, str(ex), kind="error")
                self._remove_picker(picker)
        except Exception as ex:
            snack(self.page, str(ex), kind="error")
            self._remove_picker(picker)

    def _on_backup(self, e=None):
        picker = ft.FilePicker(
            on_result=lambda ev: self._backup_result(ev, picker))
        self._add_picker(picker)
        default_name = (
            f"sierra_backup_{datetime.now().strftime('%Y%m%d')}.db")
        try:
            self.page.run_task(self._async_save, picker, default_name)
        except Exception as ex:
            snack(self.page, str(ex), kind="error")
            self._remove_picker(picker)

    def _backup_result(self, e, picker):
        try:
            path = self._extract_path(e)
            if not path:
                return
            if not path.lower().endswith(".db"):
                path = path + ".db"
            saved = backup_database(path)
            snack(self.page,
                  f"تم حفظ النسخة الاحتياطية في: {saved}",
                  kind="success")
        except Exception as ex:
            snack(self.page, str(ex), kind="error")
        finally:
            self._remove_picker(picker)

    def _on_restore(self, e=None):
        picker = ft.FilePicker(
            on_result=lambda ev: self._restore_pick_result(ev, picker))
        self._add_picker(picker)
        try:
            self.page.run_task(self._async_pick, picker)
        except Exception as ex:
            snack(self.page, str(ex), kind="error")
            self._remove_picker(picker)

    def _restore_pick_result(self, e, picker):
        self._remove_picker(picker)
        try:
            path = self._extract_path(e)
            if not path:
                return

            def do_restore():
                try:
                    restore_database_from(path)
                except Exception as ex:
                    snack(self.page, str(ex), kind="error")
                    return
                snack(self.page,
                      "تمت الاستعادة بنجاح. سيتم تسجيل الخروج.",
                      kind="success")
                try:
                    self.app.logout()
                except Exception:
                    pass

            confirm_dialog(
                self.page, "استعادة البيانات",
                f"سيتم استبدال قاعدة البيانات الحالية بالنسخة المحددة:\n"
                f"{path}\n\n"
                "ستتم إعادة تسجيل الدخول بعد الاستعادة.\n\n"
                "هل تريد المتابعة؟",
                do_restore, confirm_text="استعادة", danger=True)
        except Exception as ex:
            snack(self.page, str(ex), kind="error")

    def _on_clear_data(self, e=None):
        def do_clear():
            try:
                backup_database()
                clear_operational_data()
                snack(self.page,
                      "تم مسح الحركات وحفظ نسخة احتياطية تلقائية.",
                      kind="success")
            except Exception as ex:
                snack(self.page, str(ex), kind="error")

        confirm_dialog(
            self.page, "مسح بيانات النظام",
            "سيتم مسح جميع الحركات (فواتير، مشتريات، مصروفات، مخزون) "
            "مع أخذ نسخة احتياطية تلقائية قبل المسح. "
            "المستخدمون والموردون والمنتجات وأرقام الفواتير تبقى "
            "محفوظة. هل تريد المتابعة؟",
            do_clear, confirm_text="مسح البيانات", danger=True)

    def _update(self):
        try:
            self.container.update()
        except Exception:
            pass

NAV_ITEMS = [
    ("dashboard", "لوحة التحكم",    ft.icons.DASHBOARD,      False),
    ("pos",       "نقطة البيع",     ft.icons.POINT_OF_SALE,  False),

    ("inventory", "المخزون",         ft.icons.INVENTORY_2,    False),
    ("purchases", "المشتريات",       ft.icons.SHOPPING_BAG,   False),
    ("suppliers", "الموردون",        ft.icons.LOCAL_SHIPPING, False),
    ("expenses",  "المصروفات",       ft.icons.RECEIPT_LONG,   False),
    ("seasons",   "المواسم",         ft.icons.CALENDAR_MONTH, True),
    ("reports",   "التقارير",        ft.icons.BAR_CHART,      True),
    ("settings",  "الإعدادات",       ft.icons.SETTINGS,       True),
]


class MainLayout:
    def __init__(self, app, page, user):
        self.app = app
        self.page = page
        self.user = user
        self.is_admin = user["role"] == ROLE_ADMIN
        self.content_area = ft.Container(expand=True, padding=SP_LG)
        self.sidebar_col = ft.Column(spacing=4, expand=True,
                                     scroll=ft.ScrollMode.AUTO)
        self.current_key = None

    def build(self):
        sidebar = self._build_sidebar()
        header = self._build_header()
        body = ft.Column(
            [header,
             ft.Container(content=self.content_area, expand=True,
                          bgcolor=BG_BLACK)],
            spacing=0, expand=True)
        root = ft.Row([sidebar,
                       ft.Container(content=body, expand=True)],
                      spacing=0, expand=True)
        self._render_sidebar()
        self.navigate("dashboard")
        return root

    def _build_sidebar(self):
        brand = ft.Container(
            content=ft.Row(
                [ft.Container(
                    content=ft.Icon(ft.icons.STOREFRONT, size=18,
                                    color=BG_BLACK),
                    width=36, height=36, bgcolor=ACCENT,
                    border_radius=RADIUS_SM,
                    alignment=ft.alignment.center),
                 ft.Column(
                     [ft.Text("SIERRA", color=TEXT_PRIMARY, size=16,
                              weight=ft.FontWeight.W_900),
                      ft.Text("STORE", color=TEXT_MUTED, size=FS_TINY,
                              weight=ft.FontWeight.W_700)],
                     spacing=0, tight=True)],
                spacing=SP_SM),
            padding=ft.padding.symmetric(horizontal=SP_MD,
                                         vertical=SP_MD))
        return ft.Container(
            content=ft.Column(
                [brand, divider(),
                 ft.Container(content=self.sidebar_col, padding=SP_SM,
                              expand=True),
                 divider(), self._build_user_box()],
                spacing=0, expand=True),
            width=SIDEBAR_WIDTH, bgcolor=BG_SIDEBAR,
            border=ft.border.only(right=ft.BorderSide(1, BORDER)),
            expand=False)

    def _render_sidebar(self):
        self.sidebar_col.controls.clear()
        for key, label, icon, admin_only in NAV_ITEMS:
            if admin_only and not self.is_admin:
                continue
            self.sidebar_col.controls.append(
                self._nav_item(key, label, icon))

    def _nav_item(self, key, label, icon):
        is_active = self.current_key == key

        def _click(e, k=key):
            self.navigate(k)

        c = ft.Container(
            content=ft.Row(
                [ft.Icon(icon, size=16,
                         color=(TEXT_PRIMARY if is_active
                                else TEXT_SECONDARY)),
                 ft.Text(label, size=FS_BODY,
                         color=(TEXT_PRIMARY if is_active
                                else TEXT_SECONDARY),
                         weight=(ft.FontWeight.W_600 if is_active
                                 else ft.FontWeight.W_500))],
                spacing=SP_SM),
            padding=ft.padding.symmetric(horizontal=SP_MD, vertical=10),
            border_radius=RADIUS_SM,
            bgcolor=BG_CARD if is_active else "transparent",
            on_click=_click, ink=True)
        if not is_active:
            c.on_hover = self._make_nav_hover(key)
        return c

    def _make_nav_hover(self, key):
        def handler(e):
            if self.current_key == key:
                return
            active = e.data == "true"
            e.control.bgcolor = BG_CARD_HOVER if active else "transparent"
            e.control.update()
        return handler

    def _build_user_box(self):
        role_label = "مسؤول" if self.is_admin else "كاشير"
        return ft.Container(
            content=ft.Column(
                [ft.Row(
                    [ft.Container(
                        content=ft.Text(
                            (self.user["username"] or "?")[0].upper(),
                            color=TEXT_PRIMARY, size=FS_SMALL,
                            weight=ft.FontWeight.W_800),
                        width=32, height=32, bgcolor=BG_CARD,
                        border_radius=999,
                        alignment=ft.alignment.center),
                     ft.Column(
                         [ft.Text(self.user["username"],
                                  color=TEXT_PRIMARY, size=FS_SMALL,
                                  weight=ft.FontWeight.W_700),
                          ft.Text(role_label, color=TEXT_MUTED,
                                  size=FS_TINY)],
                         spacing=0, tight=True)],
                    spacing=SP_SM),
                 ft.Container(height=SP_SM),
                 ft.Container(
                     content=ft.Row(
                         [ft.Icon(ft.icons.LOGOUT, size=14,
                                  color=DANGER),
                          ft.Text("تسجيل الخروج", color=DANGER,
                                  size=FS_SMALL,
                                  weight=ft.FontWeight.W_600)],
                         spacing=6),
                     padding=ft.padding.symmetric(horizontal=SP_SM,
                                                  vertical=8),
                     border_radius=RADIUS_SM,
                     on_click=lambda e: self.app.logout(),
                     ink=True, on_hover=_hover_danger)],
                spacing=0),
            padding=SP_SM)

    def _build_header(self):
        now = ft.Text("", color=TEXT_SECONDARY, size=FS_SMALL)
        season_text = ft.Text("", color=SUCCESS, size=FS_SMALL,
                              weight=ft.FontWeight.W_600)

        def _tick():
            try:
                now.value = datetime.now().strftime(
                    "%A، %d %B %Y  •  %H:%M:%S")
                active = get_active_season()
                season_text.value = (f"الموسم: {active['season_name']}"
                                     if active else "")
                now.update()
                season_text.update()
            except Exception:
                pass

        _tick()
        try:
            self.page.run_task(self._clock_loop, now, season_text)
        except Exception:
            pass

        return ft.Container(
            content=ft.Row(
                [ft.Column(
                    [ft.Text("SIERRA STORE", color=TEXT_PRIMARY,
                             size=20, weight=ft.FontWeight.W_900),
                     ft.Text("نظام إدارة متجر الملابس بالجملة",
                             color=TEXT_MUTED, size=FS_TINY)],
                    spacing=0, tight=True),
                 ft.Container(expand=True),
                 season_text,
                 ft.Container(width=SP_MD),
                 ft.Container(
                     content=now,
                     padding=ft.padding.symmetric(horizontal=12,
                                                  vertical=6),
                     bgcolor=BG_CARD,
                     border=ft.border.all(1, BORDER),
                     border_radius=RADIUS_SM)],
                vertical_alignment=ft.CrossAxisAlignment.CENTER),
            padding=ft.padding.symmetric(horizontal=SP_LG, vertical=12),
            bgcolor=BG_HEADER,
            border=ft.border.only(bottom=ft.BorderSide(1, BORDER)),
            height=HEADER_HEIGHT)

    async def _clock_loop(self, target, season_target):
        import asyncio
        while True:
            await asyncio.sleep(1)
            try:
                target.value = datetime.now().strftime(
                    "%A، %d %B %Y  •  %H:%M:%S")
                active = get_active_season()
                season_target.value = (
                    f"الموسم: {active['season_name']}" if active else "")
                target.update()
                season_target.update()
            except Exception:
                return

    def navigate(self, key):
        if key == self.current_key:
            return
        for k, _l, _i, admin_only in NAV_ITEMS:
            if k == key and admin_only and not self.is_admin:
                snack(self.page,
                      "لا تملك صلاحية الوصول إلى هذا القسم.",
                      kind="error")
                return
        self.current_key = key
        self._render_sidebar()
        view = self._make_view(key)
        if view is None:
            self.content_area.content = ft.Text("الصفحة غير موجودة",
                                                color=TEXT_MUTED)
        else:
            self.content_area.content = view.build()
        try:
            self.content_area.update()
            self.sidebar_col.update()
        except Exception:
            pass

    def _make_view(self, key):
        if key == "dashboard":
            return DashboardView(self.app, self.page, self.user)
        if key == "pos":
            return POSView(self.app, self.page, self.user)
        if key == "sales":
            return SalesView(self.app, self.page, self.user)
        if key == "inventory":
            return InventoryView(self.app, self.page, self.user)
        if key == "purchases":
            return PurchasesView(self.app, self.page, self.user)
        if key == "suppliers":
            return SuppliersView(self.app, self.page, self.user)
        if key == "expenses":
            return ExpensesView(self.app, self.page, self.user)
        if key == "seasons":
            return SeasonsView(self.app, self.page, self.user)
        if key == "reports":
            return ReportsView(self.app, self.page, self.user)
        if key == "settings":
            return SettingsView(self.app, self.page, self.user)
        return None

class SierraApp:
    def __init__(self, page):
        self.page = page
        self.user = None
        self._setup_page()
        initialize_database()
        self.show_login()

    def _setup_page(self):
        self.page.title = "SIERRA STORE"
        self.page.theme_mode = ft.ThemeMode.DARK
        self.page.bgcolor = BG_BLACK
        self.page.padding = 0
        self.page.spacing = 0
        try:
            self.page.theme = ft.Theme(font_family="Cairo")
        except Exception:
            pass
        try:
            self.page.window.width = 1440
            self.page.window.height = 900
            self.page.window.min_width = 1100
            self.page.window.min_height = 700
        except Exception:
            try:
                self.page.window_width = 1440
                self.page.window_height = 900
            except Exception:
                pass

    def show_login(self):
        self.page.controls.clear()
        self.page.add(LoginView(self, self.page).build())
        self.page.update()

    def on_login_success(self, user):
        self.user = user
        self.show_main()

    def show_main(self):
        if not self.user:
            self.show_login()
            return
        self.page.controls.clear()
        self.page.add(MainLayout(self, self.page, self.user).build())
        self.page.update()

    def logout(self):
        self.user = None
        self.show_login()
